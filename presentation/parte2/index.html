Dijkstra: o que muda é só a fila de prioridade

  


template<weighted_graph G>
shortest_paths<W> dijkstra_heap(const G& g, vertex_id s)
{
  detail::check_dijkstra_input(g, s); // peso < 0 → throw
  auto sp = detail::init_shortest_paths(g, s);
  indexed_min_heap<W> heap(g.vertex_count());
  heap.push(s, 0);
  while (!heap.empty()) {
    const vertex_id u = heap.pop(); // dist[u] final
    g.for_each_edge(u, [&](vertex_id v, W w) {
      if (sp.dist[u] + w < sp.dist[v]) {   // estrita
        sp.dist[v] = sp.dist[u] + w; sp.parent[v] = u;
        heap.contains(v) ? heap.update_key(v, …)
                         : heap.push(v, …);
      }
    });
  }
  return sp;  // {source, dist, parent}
}
